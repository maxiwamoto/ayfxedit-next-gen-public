cmake_minimum_required(VERSION 3.21)
project(AYFXEditNextGen VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/embed_runtime.py"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia Concurrent)
qt_standard_project_setup()
set(CMAKE_AUTOUIC OFF)
qt_add_executable(ayfxedit-next-gen MACOSX_BUNDLE portable/main.cpp portable/workshop_dialog.cpp)
target_include_directories(ayfxedit-next-gen PRIVATE native portable)
target_link_libraries(ayfxedit-next-gen PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::Concurrent)
set_target_properties(ayfxedit-next-gen PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "me.maxiwamoto.ayfxedit"
    MACOSX_BUNDLE_BUNDLE_NAME "AYFXEDIT NEXT GEN"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(ayfxedit-next-gen PRIVATE /utf-8)
    target_compile_definitions(ayfxedit-next-gen PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_options(ayfxedit-next-gen PRIVATE /STACK:16777216)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(ayfx-core-tests native/tests.cpp)
    target_include_directories(ayfx-core-tests PRIVATE native)
    if(MSVC)
        target_compile_definitions(ayfx-core-tests PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
        target_link_options(ayfx-core-tests PRIVATE /STACK:16777216)
    endif()
    add_test(NAME fixtures COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_test_fixtures.py")
    set_tests_properties(fixtures PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" FIXTURES_SETUP fixture_bank)
    add_test(NAME core COMMAND ayfx-core-tests)
    set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" FIXTURES_REQUIRED fixture_bank)
    add_test(NAME portable-ui COMMAND ayfxedit-next-gen --self-test "${CMAKE_BINARY_DIR}/self-test")
    if(NOT WIN32)
        set_tests_properties(portable-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
install(TARGETS ayfxedit-next-gen BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES CREDITS.md THIRD_PARTY_NOTICES.md README.md docs/NATIVE_PORT.md DESTINATION share/ayfxedit-next-gen)

set(CPACK_PACKAGE_NAME "ayfxedit-next-gen")
set(CPACK_PACKAGE_VERSION "1.1.0")
set(CPACK_PACKAGE_CONTACT "https://github.com/maxiwamoto/ayfxedit-next-gen-public")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "AY sound-effect editor - native desktop preview")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6multimedia6, gstreamer1.0-plugins-good")
set(CPACK_GENERATOR "TGZ")
if(UNIX AND NOT APPLE)
    set(CPACK_GENERATOR "DEB;TGZ")
endif()
include(CPack)

install(DIRECTORY licenses/ DESTINATION share/ayfxedit-next-gen/licenses)
